# Ungleichungen
$a < b$ => "kleiner" => $1 < 2$
$a \leq b$ => "kleiner-gleich" => $1 \leq 2 \quad 2 \leq 2$
# Rechenregeln
$a < b, \quad c \in \mathbb{R} \implies a + c < b + c$

$a < b, \quad c \in \mathbb{R}$
- $a \cdot c < b \cdot c, \quad c > 0$
- $a \cdot c > b \cdot c, \quad c < 0$
- $a \cdot c = b \cdot c, \quad c = 0$
# Beträge
$|a| \geq 0,\quad (|a| = 0, \quad \text{nur für } a = 0)$

$|a \cdot b| = |a| \cdot |b|$

$|a+b| \leq |a| + |b|$
